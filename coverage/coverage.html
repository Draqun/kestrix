// Package main is the metaci CLI entry point.
//
// Usage:
//
//        metaci resolve <path-to-pipeline.tf> --changed auth_service,docs
//        metaci validate <path-to-pipeline.tf>
//
// "resolve" reads the HCL pipeline, evaluates trigger expressions against the
// given changed-repo set, and prints which tasks should run.
//
// "validate" parses the HCL and checks the DAG for cycles but does not require
// any changed-set input.
package main

import (
        "flag"
        "fmt"
        "os"
        "strings"

        "gitlab.com/lunaris-bot/metaci/internal/dag"
        "gitlab.com/lunaris-bot/metaci/internal/parser"
        "gitlab.com/lunaris-bot/metaci/internal/web"
)

func main() {
        if len(os.Args) < 2 {
                usage()
                os.Exit(1)
        }
        switch os.Args[1] {
        case "resolve":
                cmdResolve(os.Args[2:])
        case "validate":
                cmdValidate(os.Args[2:])
        case "serve":
                cmdServe(os.Args[2:])
        case "help", "-h", "--help":
                usage()
        default:
                fmt.Fprintf(os.Stderr, "unknown command: %s\n", os.Args[1])
                usage()
                os.Exit(2)
        }
}

func usage() {
        fmt.Fprintln(os.Stderr, `metaci — repo-agnostic CI/CD orchestrator

Usage:
  metaci resolve  [--changed repo1,repo2] <file.tf>
  metaci validate  <file.tf>
  metaci serve     [--addr :8080] [--dir ./examples]`)
}

func cmdServe(args []string) {
        fs := flag.NewFlagSet("serve", flag.ExitOnError)
        addr := fs.String("addr", ":8080", "listen address (e.g. :8080, 0.0.0.0:9000)")
        dir := fs.String("dir", "./examples", "pipeline directory containing .tf files")
        if err := fs.Parse(args); err != nil {
                os.Exit(2)
        }

        cfg := web.Config{
                ListenAddr:  *addr,
                PipelineDir: *dir,
        }
        server := web.NewServer(cfg)
        if err := server.Run(); err != nil {
                fmt.Fprintf(os.Stderr, "server error: %v\n", err)
                os.Exit(1)
        }
}

func cmdResolve(args []string) {
        fs := flag.NewFlagSet("resolve", flag.ExitOnError)
        changedFlag := fs.String("changed", "", "comma-separated list of changed repo names")
        if err := fs.Parse(args); err != nil {
                os.Exit(2)
        }
        if fs.NArg() < 1 {
                fmt.Fprintln(os.Stderr, "error: pipeline file required")
                os.Exit(2)
        }
        path := fs.Arg(0)

        proj, err := parser.ParseFile(path)
        if err != nil {
                fmt.Fprintf(os.Stderr, "parse error: %v\n", err)
                os.Exit(1)
        }

        // Build the changed-repos map: every declared repo is present; changed ones set to true.
        changed := make(map[string]bool)
        for _, r := range proj.Repos {
                changed[r.Name] = false
        }
        if *changedFlag != "" {
                for _, name := range strings.Split(*changedFlag, ",") {
                        name = strings.TrimSpace(name)
                        if _, ok := changed[name]; !ok {
                                fmt.Fprintf(os.Stderr, "warning: changed repo %q is not declared in pipeline\n", name)
                        }
                        changed[name] = true
                }
        }

        engine := dag.New(proj)
        nodes, err := engine.Resolve(changed)
        if err != nil {
                fmt.Fprintf(os.Stderr, "resolve error: %v\n", err)
                os.Exit(1)
        }

        fmt.Printf("Project: %s\n", proj.Name)
        fmt.Printf("Changed repos: %s\n", changedSummary(changed))
        fmt.Printf("Tasks triggered (%d):\n", len(nodes))
        for _, n := range nodes {
                tag := "AUTO"
                if n.Task.Manual {
                        tag = "MANUAL"
                }
                fmt.Printf("  [%s] %s\n", tag, n.Task.Name)
                fmt.Printf("    trigger: %s\n", n.Task.Trigger)
                fmt.Printf("    repo: %s | script: %s\n", n.Task.Execute.Repo, n.Task.Execute.Script)
                fmt.Printf("    reason: %s\n", n.TriggeredReason)
                if len(n.Task.
                        DependsOn) > 0 {
                        fmt.Printf("    depends_on: %s\n", strings.Join(n.Task.DependsOn, ", "))
                }
        }
}

func cmdValidate(args []string) {
        fs := flag.NewFlagSet("validate", flag.ExitOnError)
        if err := fs.Parse(args); err != nil {
                os.Exit(2)
        }
        if fs.NArg() < 1 {
                fmt.Fprintln(os.Stderr, "error: pipeline file required")
                os.Exit(2)
        }
        path := fs.Arg(0)

        proj, err := parser.ParseFile(path)
        if err != nil {
                fmt.Fprintf(os.Stderr, "parse error: %v\n", err)
                os.Exit(1)
        }

        engine := dag.New(proj)
        if err := engine.Validate(); err != nil {
                fmt.Fprintf(os.Stderr, "validation failed: %v\n", err)
                os.Exit(1)
        }

        fmt.Printf("OK — project %q validated\n", proj.Name)
        fmt.Printf("  repos: %d\n", len(proj.Repos))
        fmt.Printf("  tasks: %d\n", len(proj.Tasks))
}

func changedSummary(changed map[string]bool) string {
        var parts []string
        for name, v := range changed {
                if v {
                        parts = append(parts, name)
                }
        }
        if len(parts) == 0 {
                return "(none)"
        }
        return strings.Join(parts, ", ")
}

// Package dag implements the dependency-graph engine for MetaCI.
//
// The engine takes a parsed model.Project and a set of "changed repos" (derived
// from incoming webhook events or git diffs) and computes which tasks should
// run, in what order, and which are blocked by manual-approval gates.
package dag

import (
        "fmt"
        "strings"

        "gitlab.com/lunaris-bot/metaci/internal/model"
)

// Node is a single task in the resolved DAG, annotated with whether its trigger
// condition is satisfied and whether it requires manual approval.
type Node struct {
        Task            model.Task
        Triggered       bool
        TriggeredReason string
}

// Engine holds the parsed project and provides operations to evaluate triggers
// and produce an execution plan.
type Engine struct {
        Project model.Project
}

// New creates a DAG engine from a parsed project.
func New(p model.Project) *Engine {
        return &Engine{Project: p}
}

// EvaluateTrigger computes whether a task's trigger expression is satisfied given
// the set of changed repo names. It parses the boolean expression supporting
// AND/OR/NOT and the "repo.<name>.changed" predicate.
//
// Example expressions accepted:
//
//        repo.auth_service.changed && !repo.docs.changed
//        (repo.a.changed || repo.b.changed) && !repo.docs.changed
func (e *Engine) EvaluateTrigger(expr string, changed map[string]bool) (bool, error) {
        tokens, err := tokenize(expr)
        if err != nil {
                return false, fmt.Errorf("tokenize: %w", err)
        }
        parser := &exprParser{tokens: tokens}
        result, err := parser.parseExpr(changed)
        if err != nil {
                return false, fmt.Errorf("evaluate trigger %q: %w", expr, err)
        }
        return result, nil
}

// Resolve computes the full execution plan. It evaluates every task's trigger
// condition and returns only the nodes that should run (triggered == true).
// Ordering via depends_on is validated for cycles.
func (e *Engine) Resolve(changed map[string]bool) ([]Node, error) {
        var nodes []Node
        for _, task := range e.Project.Tasks {
                ok, err := e.EvaluateTrigger(task.Trigger, changed)
                if err != nil {
                        return nil, err
                }
                n := Node{Task: task, Triggered: ok}
                if ok {
                        n.TriggeredReason = "trigger expression satisfied"
                } else {
                        n.TriggeredReason = "trigger expression not satisfied"
                }
                if ok {
                        nodes = append(nodes, n)
                }
        }
        // Validate dependency ordering (cycle detection).
        if err := e.detectCycles(); err != nil {
                return nil, err
        }
        return nodes, nil
}

// Validate checks the project structure: trigger syntax and DAG cycles.
// It does not require a changed-set; it validates the configuration itself.
func (e *Engine) Validate() error {
        // Syntax-check all trigger expressions with an empty changed set.
        empty := make(map[string]bool)
        for _, r := range e.Project.Repos {
                empty[r.Name] = false
        }
        for _, t := range e.Project.Tasks {
                if _, err := e.EvaluateTrigger(t.Trigger, empty); err != nil {
                        return fmt.Errorf("task %q: %w", t.Name, err)
                }
        }
        // Check for cross-repo execution references: every execute.repo must
        // resolve to a declared ci_repo.
        repoNames := make(map[string]bool)
        for _, r := range e.Project.Repos {
                repoNames[r.Name] = true
        }
        for _, t := range e.Project.Tasks {
                if !repoNames[t.Execute.Repo] {
                        return fmt.Errorf("task %q: execute.repo %q is not a declared ci_repo", t.Name, t.Execute.Repo)
                }
        }
        return e.detectCycles()
}

// detectCycles performs a depth-first traversal of depends_on edges and returns an
// error if a cycle is found.
func (e *Engine) detectCycles() error {
        adj := make(map[string][]string)
        taskNames := make(map[string]bool)
        for _, t := range e.Project.Tasks {
                taskNames[t.Name] = true
                adj[t.Name] = t.DependsOn
        }
        visited := make(map[string]int) // 0=unvisited,1=in-stack,2=done
        var dfs func(name string, path []string) error
        dfs = func(name string, path []string) error {
                if visited[name] == 1 {
                        return fmt.Errorf("cycle detected: %s -> %s", strings.Join(path, " -> "), name)
                }
                if visited[name] == 2 {
                        return nil
                }
                if !taskNames[name] {
                        return fmt.Errorf("unknown dependency: %q", name)
                }
                visited[name] = 1
                for _, dep := range adj[name] {
                        if err := dfs(dep, append(path, name)); err != nil {
                                return err
                        }
                }
                visited[name] = 2
                return nil
        }
        for name := range taskNames {
                if visited[name] == 0 {
                        if err := dfs(name, nil); err != nil {
                                return err
                        }
                }
        }
        return nil
}

// Package dag — expression parser for trigger conditions.
//
// Supports a minimal grammar sufficient for CI trigger logic:
//
//        expr   := orExpr
//        orExpr := andExpr ( "||" andExpr )*
//        andExpr:= notExpr ( "&&" notExpr )*
//        notExpr:= "!" notExpr | primary
//        primary:= "(" expr ")" | predicate
//
// predicate := "repo" "." <name> "." "changed"
//
// Tokens are identifiers, dots, parentheses, and the operators && || !.
package dag

import (
        "fmt"
        "strings"
        "unicode"
)

type tokenKind int

const (
        tkIdent tokenKind = iota
        tkDot
        tkAnd // &&
        tkOr  // ||
        tkNot // !
        tkLParen
        tkRParen
        tkEOF
)

type token struct {
        kind tokenKind
        text string
}

func tokenize(s string) ([]token, error) {
        var tokens []token
        i := 0
        for i < len(s) {
                ch := s[i]
                switch {
                case ch == ' ' || ch == '\t' || ch == '\n':
                        i++
                case ch == '.':
                        tokens = append(tokens, token{tkDot, "."})
                        i++
                case ch == '(':
                        tokens = append(tokens, token{tkLParen, "("})
                        i++
                case ch == ')':
                        tokens = append(tokens, token{tkRParen, ")"})
                        i++
                case ch == '!' && i+1 < len(s) && s[i+1] == '=':
                        return nil, fmt.Errorf("invalid token '!=' (use '!' for negation) at position %d", i)
                case ch == '!':
                        tokens = append(tokens, token{tkNot, "!"})
                        i++
                case ch == '&' && i+1 < len(s) && s[i+1] == '&':
                        tokens = append(tokens, token{tkAnd, "&&"})
                        i += 2
                case ch == '|' && i+1 < len(s) && s[i+1] == '|':
                        tokens = append(tokens, token{tkOr, "||"})
                        i += 2
                case isIdentRune(rune(ch)):
                        start := i
                        for i < len(s) && isIdentRune(rune(s[i])) {
                                i++
                        }
                        tokens = append(tokens, token{tkIdent, s[start:i]})
                default:
                        return nil, fmt.Errorf("unexpected char %q at position %d", ch, i)
                }
        }
        tokens = append(tokens, token{tkEOF, ""})
        return tokens, nil
}

func isIdentRune(r rune) bool {
        return r == '_' || r == '-' || unicode.IsLetter(r) || unicode.IsDigit(r)
}

type exprParser struct {
        tokens []token
        pos    int
}

func (p *exprParser) peek() token {
        if p.pos < len(p.tokens) {
                return p.tokens[p.pos]
        }
        return token{tkEOF, ""}
}

func (p *exprParser) advance() token {
        t := p.peek()
        p.pos++
        return t
}

func (p *exprParser) parseExpr(changed map[string]bool) (bool, error) {
        return p.parseOr(changed)
}

func (p *exprParser) parseOr(changed map[string]bool) (bool, error) {
        left, err := p.parseAnd(changed)
        if err != nil {
                return false, err
        }
        for p.peek().kind == tkOr {
                p.advance()
                right, err := p.parseAnd(changed)
                if err != nil {
                        return false, err
                }
                left = left || right
        }
        return left, nil
}

func (p *exprParser) parseAnd(changed map[string]bool) (bool, error) {
        left, err := p.parseNot(changed)
        if err != nil {
                return false, err
        }
        for p.peek().kind == tkAnd {
                p.advance()
                right, err := p.parseNot(changed)
                if err != nil {
                        return false, err
                }
                left = left && right
        }
        return left, nil
}

func (p *exprParser) parseNot(changed map[string]bool) (bool, error) {
        if p.peek().kind == tkNot {
                p.advance()
                val, err := p.parseNot(changed)
                if err != nil {
                        return false, err
                }
                return !val, nil
        }
        return p.parsePrimary(changed)
}

func (p *exprParser) parsePrimary(changed map[string]bool) (bool, error) {
        t := p.peek()
        switch t.kind {
        case tkLParen:
                p.advance()
                val, err := p.parseExpr(changed)
                if err != nil {
                        return false, err
                }
                if p.peek().kind != tkRParen {
                        return false, fmt.Errorf("expected ')' got %q", p.peek().text)
                }
                p.advance()
                return val, nil
        case tkIdent:
                return p.parsePredicate(changed)
        default:
                return false, fmt.Errorf("unexpected token %q", t.text)
        }
}

// parsePredicate handles "repo.<name>.changed"
func (p *exprParser) parsePredicate(changed map[string]bool) (bool, error) {
        // repo
        t := p.advance()
        if t.text != "repo" {
                return false, fmt.Errorf("expected 'repo' got %q", t.text)
        }
        // .
        if p.peek().kind != tkDot {
                return false, fmt.Errorf("expected '.' after 'repo'")
        }
        p.advance()
        // <name>
        nameTok := p.advance()
        if nameTok.kind != tkIdent {
                return false, fmt.Errorf("expected repo name after 'repo.'")
        }
        // .
        if p.peek().kind != tkDot {
                return false, fmt.Errorf("expected '.' after repo name %q", nameTok.text)
        }
        p.advance()
        // changed
        propTok := p.advance()
        if propTok.kind != tkIdent || propTok.text != "changed" {
                return false, fmt.Errorf("expected 'changed' got %q", propTok.text)
        }
        repoName := nameTok.text
        _, exists := changed[repoName]
        if !exists {
                return false, fmt.Errorf("unknown repo %q in trigger (not declared as ci_repo)", repoName)
        }
        return changed[repoName], nil
}

// String returns a human-readable label for the changed set — useful for logging.
func changedString(changed map[string]bool) string {
        var parts []string
        for name, v := range changed {
                if v {
                        parts = append(parts, name)
                }
        }
        return strings.Join(parts, ", ")
}

// Package parser reads HCL files and produces model.Project values.
//
// We deliberately avoid gohcl: it requires an *hcl.EvalContext even though our
// HCL contains only literal values (no interpolation, no variables). Manual
// decoding via Body.Blocks/JustAttributes is simpler and has no EvalContext
// requirement.
package parser

import (
        "fmt"
        "os"
        "path/filepath"
        "strings"

        "github.com/hashicorp/hcl/v2"
        "github.com/hashicorp/hcl/v2/hclsyntax"
        "github.com/zclconf/go-cty/cty"

        "gitlab.com/lunaris-bot/metaci/internal/model"
)

// ParseFile reads a single HCL file and returns its Project definition.
func ParseFile(path string) (model.Project, error) {
        src, err := os.ReadFile(path)
        if err != nil {
                return model.Project{}, fmt.Errorf("read %s: %w", path, err)
        }
        return ParseSource(src, path)
}

// ParseSource parses HCL source bytes into a Project.
func ParseSource(src []byte, filename string) (model.Project, error) {
        f, diags := hclsyntax.ParseConfig(src, filename, hcl.Pos{Line: 1, Column: 1})
        if diags.HasErrors() {
                return model.Project{}, formatDiags(diags, filename)
        }
        body := f.Body.(*hclsyntax.Body)

        var proj model.Project
        for _, blk := range body.Blocks {
                switch blk.Type {
                case "project":
                        p, err := decodeProjectBlock(blk)
                        if err != nil {
                                return model.Project{}, fmt.Errorf("project block: %w", err)
                        }
                        proj = p
                default:
                        // Top-level blocks must be `project`.
                        return model.Project{}, fmt.Errorf("unsupported top-level block %q", blk.Type)
                }
        }
        return proj, nil
}

// decodeProjectBlock reads the inner attributes (name, default_image) and
// child blocks (ci_repo, ci_task) of a project block.
func decodeProjectBlock(blk *hclsyntax.Block) (model.Project, error) {
        if len(blk.Labels) != 1 {
                return model.Project{}, fmt.Errorf("project block requires exactly 1 label, got %d", len(blk.Labels))
        }
        proj := model.Project{Name: blk.Labels[0]}

        // Iterate ALL top-level items of the body so we get both attributes and
        // blocks, delegating each to the proper extractor.
        for _, item := range blk.Body.Attributes {
                val, err := literal(item.Expr)
                if err != nil {
                        return model.Project{}, fmt.Errorf("attribute %q: %w", item.Name, err)
                }
                switch item.Name {
                case "default_image":
                        proj.DefaultImage = val
                }
        }

        for _, child := range blk.Body.Blocks {
                switch child.Type {
                case "ci_repo":
                        r, err := decodeRepoBlock(child)
                        if err != nil {
                                return model.Project{}, fmt.Errorf("ci_repo: %w", err)
                        }
                        proj.Repos = append(proj.Repos, r)
                case "ci_task":
                        t, err := decodeTaskBlock(child)
                        if err != nil {
                                return model.Project{}, fmt.Errorf("ci_task: %w", err)
                        }
                        proj.Tasks = append(proj.Tasks, t)
                default:
                        return model.Project{}, fmt.Errorf("unsupported child block %q in project", child.Type)
                }
        }
        return proj, nil
}

func decodeRepoBlock(blk *hclsyntax.Block) (model.Repo, error) {
        if len(blk.Labels) != 1 {
                return model.Repo{}, fmt.Errorf("ci_repo requires 1 label")
        }
        r := model.Repo{Name: blk.Labels[0]}
        for name, attr := range attrsMap(blk.Body) {
                val, err := literal(attr.Expr)
                if err != nil {
                        return model.Repo{}, fmt.Errorf("attribute %q: %w", name, err)
                }
                switch name {
                case "path":
                        r.Path = val
                case "type":
                        r.Type = val
                }
        }
        return r, nil
}

func decodeTaskBlock(blk *hclsyntax.Block) (model.Task, error) {
        if len(blk.Labels) != 1 {
                return model.Task{}, fmt.Errorf("ci_task requires 1 label")
        }
        t := model.Task{Name: blk.Labels[0]}

        // Top-level attributes of ci_task: trigger, manual, depends_on
        for name, attr := range attrsMap(blk.Body) {
                val, err := literal(attr.Expr)
                if err != nil {
                        return model.Task{}, fmt.Errorf("attribute %q: %w", name, err)
                }
                switch name {
                case "trigger":
                        t.Trigger = val
                case "manual":
                        t.Manual = val == "true"
                case "depends_on":
                        t.DependsOn = splitList(val)
                }
        }

        // execute block (optional, at most one)
        for _, child := range blk.Body.Blocks {
                switch child.Type {
                case "execute":
                        t.Execute = model.ExecuteSpec{}
                        for name, attr := range attrsMap(child.Body) {
                                val, err := literal(attr.Expr)
                                if err != nil {
                                        return model.Task{}, fmt.Errorf("execute.%s: %w", name, err)
                                }
                                switch name {
                                case "repo":
                                        t.Execute.Repo = val
                                case "script":
                                        t.Execute.Script = val
                                case "image":
                                        t.Execute.Image = val
                                }
                        }
                default:
                        return model.Task{}, fmt.Errorf("unsupported child block %q in ci_task", child.Type)
                }
        }
        return t, nil
}

// attrsMap returns the block body's attributes as a name -> *Attribute map by
// reading the hclsyntax.Body.Attributes slice directly. We don't use
// JustAttributes because it errors out if the body contains any sub-blocks
// (which our bodies do).
func attrsMap(body *hclsyntax.Body) map[string]*hclsyntax.Attribute {
        out := map[string]*hclsyntax.Attribute{}
        for _, a := range body.Attributes {
                out[a.Name] = a
        }
        return out
}

// literal extracts the string value of a literal expression. We support:
//   - string literals (e.g.  "foo")
//   - tuple expressions (lists of literals), comma-separated
//   - bool literals
//
// We deliberately do NOT support variable interpolation (functions, refs),
// because pipeline configuration should be declarative and reproducible.
func literal(expr hcl.Expression) (string, error) {
        switch e := expr.(type) {
        case *hclsyntax.TemplateExpr:
                if len(e.Parts) == 1 {
                        if ts, ok := e.Parts[0].(*hclsyntax.LiteralValueExpr); ok {
                                return toString(ts.Val)
                        }
                }
                return "", fmt.Errorf("interpolated strings are not supported in pipeline config")
        case *hclsyntax.LiteralValueExpr:
                return toString(e.Val)
        case *hclsyntax.TupleConsExpr:
                parts, err := tupleStrings(e)
                if err != nil {
                        return "", err
                }
                return strings.Join(parts, ","), nil
        case *hclsyntax.ObjectConsExpr:
                return "", fmt.Errorf("object literals are not supported here")
        case *hclsyntax.SplatExpr, *hclsyntax.RelativeTraversalExpr, *hclsyntax.ScopeTraversalExpr:
                return "", fmt.Errorf("variable/scoped references are not supported; use literal values only")
        default:
                return "", fmt.Errorf("unsupported expression type %T", expr)
        }
}

func tupleStrings(e *hclsyntax.TupleConsExpr) ([]string, error) {
        out := make([]string, 0, len(e.Exprs))
        for _, ex := range e.Exprs {
                s, err := literal(ex)
                if err != nil {
                        return nil, err
                }
                out = append(out, s)
        }
        return out, nil
}

func toString(v interface{}) (string, error) {
        // hclsyntax.LiteralValueExpr.Val is a cty.Value.
        switch x := v.(type) {
        case cty.Value:
                if x.Type() == cty.String {
                        return x.AsString(), nil
                }
                if x.Type() == cty.Bool {
                        if x.True() {
                                return "true", nil
                        }
                        return "false", nil
                }
                return "", fmt.Errorf("unsupported cty type %s", x.Type().FriendlyName())
        case string:
                return x, nil
        case bool:
                if x {
                        return "true", nil
                }
                return "false", nil
        default:
                return "", fmt.Errorf("unsupported literal type %T", v)
        }
}

func splitList(s string) []string {
        parts := strings.Split(s, ",")
        out := make([]string, 0, len(parts))
        for _, p := range parts {
                p = strings.TrimSpace(p)
                if p != "" {
                        out = append(out, p)
                }
        }
        return out
}

type executeZero = model.ExecuteSpec

// ParseDir reads all .tf files in a directory and merges them into one Project.
func ParseDir(dir string) (model.Project, error) {
        entries, err := os.ReadDir(dir)
        if err != nil {
                return model.Project{}, fmt.Errorf("read dir %s: %w", dir, err)
        }
        var proj model.Project
        for _, e := range entries {
                if e.IsDir() || !strings.HasSuffix(e.Name(), ".tf") {
                        continue
                }
                path := filepath.Join(dir, e.Name())
                p, err := ParseFile(path)
                if err != nil {
                        return model.Project{}, fmt.Errorf("parse %s: %w", path, err)
                }
                proj.Repos = append(proj.Repos, p.Repos...)
                proj.Tasks = append(proj.Tasks, p.Tasks...)
                if proj.Name == "" && p.Name != "" {
                        proj.Name = p.Name
                }
                if p.DefaultImage != "" {
                        proj.DefaultImage = p.DefaultImage
                }
        }
        return proj, nil
}

func formatDiags(diags hcl.Diagnostics, filename string) error {
        var sb strings.Builder
        for _, d := range diags {
                sb.WriteString(fmt.Sprintf("[%v] %s:%d:%d: %s\n",
                        d.Severity, filename, d.Subject.Start.Line, d.Subject.Start.Column, d.Summary))
        }
        return fmt.Errorf("%s", sb.String())
}

import (

        "strings"

        "gitlab.com/lunaris-bot/metaci/internal/dag"
        "gitlab.com/lunaris-bot/metaci/internal/model"
)

// GenerateGitLabCI produces a .gitlab-ci.yml string from the resolved DAG nodes.
//
// The generated YAML is minimal but functional: each triggered task becomes a
// job. Dependencies are mapped to GitLab's "needs" keyword. Manual tasks get
// "when: manual". Container images are set per-job (the per-job dynamic image
// pulling concept).
func GenerateGitLabCI(proj model.Project, nodes []dag.Node) string {
        var sb strings.Builder

        // Header comment.
        sb.WriteString("# Auto-generated by MetaCI — do not edit manually\n")
        sb.WriteString(fmt.Sprintf("# Project: %s\n\n", proj.Name))

        // Stages: derive from the dependency order.
        // Simple approach: all jobs start in "build" unless they depend on something,
        // then they go to "test", and if they depend on test jobs, "deploy".
        stages := deriveStages(nodes)
        if len(stages) > 0 {
                sb.WriteString("stages:\n")
                for _, s := range stages {
                        sb.WriteString(fmt.Sprintf("  - %s\n", s))
                }
                sb.WriteString("\n")
        }

        // Default image (from project-level setting).
        if proj.DefaultImage != "" {
                sb.WriteString(fmt.Sprintf("default:\n  image: %s\n\n", proj.DefaultImage))
        }

        // Jobs.
        for _, node := range nodes {
                job := node.Task
                stage := assignStage(job, nodes)

                sb.WriteString(fmt.Sprintf("%s:\n", job.Name))
                sb.WriteString(fmt.Sprintf("  stage: %s\n", stage))

                // Image: per-job override or project default.
                if job.Execute.Image != "" {
                        sb.WriteString(fmt.Sprintf("  image: %s\n", job.Execute.Image))
                }

                // Dependencies.
                if len(job.DependsOn) > 0 {
                        sb.WriteString("  needs:\n")
                        for _, dep := range job.DependsOn {
                                sb.WriteString(fmt.Sprintf("    - %s\n", dep))
                        }
                }

                // Manual approval gate.
                if job.Manual {
                        sb.WriteString("  when: manual\n")
                }

                // Script: clone the repo and run the script.
                sb.WriteString("  script:\n")
                sb.WriteString(fmt.Sprintf("    - echo \"Running %s in repo %s\"\n", job.Name, job.Execute.Repo))
                sb.WriteString(fmt.Sprintf("    - %s\n", job.Execute.Script))

                sb.WriteString("\n")
        }

        return sb.String()
}

// deriveStages computes the pipeline stages from the node dependency graph.
func deriveStages(nodes []dag.Node) []string {
        hasNoDeps := false
        hasTestDeps := false
        hasDeployDeps := false

        for _, n := range nodes {
                if len(n.Task.DependsOn) == 0 {
                        hasNoDeps = true
                }
                for _, dep := range n.Task.DependsOn {
                        // Check if the dependency is a "test" job.
                        if strings.Contains(dep, "test") {
                                hasTestDeps = true
                        }
                }
                if n.Task.Manual {
                        hasDeployDeps = true
                }
        }

        var stages []string
        if hasNoDeps {
                stages = append(stages, "build")
        }
        if hasTestDeps {
                stages = append(stages, "test")
        }
        if hasDeployDeps {
                stages = append(stages, "deploy")
        }
        if len(stages) == 0 {
                stages = append(stages, "build")
        }
        return stages
}

// assignStage determines which stage a task belongs to.
func assignStage(task model.Task, nodes []dag.Node) string {
        if task.Manual {
                return "deploy"
        }
        // If the task has dependencies, it's at least "test" level.
        if len(task.DependsOn) > 0 {
                for _, dep := range task.DependsOn {
                        if strings.Contains(dep, "test") {
                                return "deploy"
                        }
                }
                return "test"
        }
        return "build"
}

import (

// Config holds the server configuration.
type Config struct {
        // ListenAddr is the address to bind the HTTP server to (e.g. ":8080").
        ListenAddr string

        // PipelineDir is the directory containing the pipeline .tf files.
        PipelineDir string
}

// Server is the MetaCI webhook receiver. It listens for GitLab push events,
// evaluates the DAG against the changed files, and dispatches tasks.
type Server struct {
        cfg        Config
        classifier *FileClassifier
}

// NewServer creates a new webhook server.
func NewServer(cfg Config) *Server {
        return &Server{
                cfg:        cfg,
                classifier: DefaultClassifier(),
        }
}

// Router returns the Chi router with all routes registered.
func (s *Server) Router() *chi.Mux {
        r := chi.NewRouter()

        // Middleware stack.
        r.Use(middleware.Logger)
        r.Use(middleware.Recoverer)
        r.Use(middleware.RequestID)
        r.Use(middleware.Heartbeat("/ping"))

        // Routes.
        r.Get("/health", s.handleHealth)
        r.Post("/webhook", s.handleWebhook)
        r.Get("/", s.handleRoot)

        return r
}

// Run starts the HTTP server and blocks until it returns an error.
func (s *Server) Run() error {
        router := s.Router()
        log.Printf("MetaCI webhook server listening on %s", s.cfg.ListenAddr)
        log.Printf("Pipeline dir: %s", s.cfg.PipelineDir)
        return http.ListenAndServe(s.cfg.ListenAddr, router)
}

// --- Handlers ---

func (s *Server) handleRoot(w http.ResponseWriter, r *http.Request) {
        w.Header().Set("Content-Type", "application/json")
        json.NewEncoder(w).Encode(map[string]interface{}{
                "name":    "MetaCI",
                "version": "0.1.0",
                "status":  "running",
                "endpoints": map[string]string{
                        "webhook": "POST /webhook",
                        "health":  "GET /health",
                        "ping":    "GET /ping",
                },
        })
}

func (s *Server) handleHealth(w http.ResponseWriter, r *http.Request) {
        w.Header().Set("Content-Type", "application/json")
        json.NewEncoder(w).Encode(map[string]string{"status": "ok"})
}

// handleWebhook is the main webhook handler.
func (s *Server) handleWebhook(w http.ResponseWriter, r *http.Request) {
        body, err := io.ReadAll(r.Body)
        if err != nil {
                s.writeError(w, http.StatusBadRequest, fmt.Sprintf("read body: %v", err))
                return
        }
        defer r.Body.Close()

        // Step 1: Parse the webhook payload.
        event, err := ParseGitLabPushEvent(body)
        if err != nil {
                log.Printf("webhook parse error: %v", err)
                s.writeError(w, http.StatusBadRequest, fmt.Sprintf("parse error: %v", err))
                return
        }

        log.Printf("webhook: repo=%s branch=%s commits=%d files=%d",
                event.Repo, event.Branch, event.CommitCount, len(event.FilesChanged))

        if event.IsEmpty() {
                log.Printf("no file changes, skipping")
                s.writeJSON(w, http.StatusOK, map[string]interface{}{
                        "status": "skipped",
                        "reason": "no file changes in push",
                })
                return
        }

        // Step 2: Load the pipeline definition.
        proj, err := parser.ParseDir(s.cfg.PipelineDir)
        if err != nil {
                log.Printf("pipeline parse error: %v", err)
                s.writeError(w, http.StatusInternalServerError, fmt.Sprintf("pipeline parse error: %v", err))
                return
        }

        // Step 3: Classify changes.
        changedRepos := s.classifyChanges(event, proj)

        // Step 4: Evaluate DAG.
        engine := dag.New(proj)
        nodes, err := engine.Resolve(changedRepos)
        if err != nil {
                log.Printf("DAG resolve error: %v", err)
                s.writeError(w, http.StatusInternalServerError, fmt.Sprintf("DAG error: %v", err))
                return
        }

        log.Printf("resolved %d tasks to execute", len(nodes))

        // Step 5: Generate gitlab-ci.yml.
        yamlContent := GenerateGitLabCI(proj, nodes)

        // Step 6: Return the result.
        s.writeJSON(w, http.StatusOK, map[string]interface{}{
                "status":        "ok",
                "project":       proj.Name,
                "repo":          event.Repo,
                "branch":        event.Branch,
                "changed_repos": changedRepos,
                "tasks":         nodesToJSON(nodes),
                "gitlab_ci":     yamlContent,
        })
}

// --- Helpers ---

func (s *Server) classifyChanges(event *PushEvent, proj model.Project) map[string]bool {
        repoPaths := make(map[string]string)
        for _, r := range proj.Repos {
                repoPaths[r.Name] = r.Path
        }

        changed := make(map[string]bool)
        for name, path := range repoPaths {
                if path == event.Repo || event.Repo == path {
                        changed[name] = s.classifier.HasCodeChanges(event.FilesChanged)
                } else {
                        changed[name] = false
                }
        }

        if len(changed) == 0 {
                log.Printf("warning: repo %q not found in pipeline project %q", event.Repo, proj.Name)
        }

        return changed
}

func (s *Server) writeJSON(w http.ResponseWriter, status int, v interface{}) {
        w.Header().Set("Content-Type", "application/json")
        w.WriteHeader(status)
        json.NewEncoder(w).Encode(v)
}

func (s *Server) writeError(w http.ResponseWriter, status int, msg string) {
        log.Printf("error %d: %s", status, msg)
        s.writeJSON(w, status, map[string]string{"error": msg})
}

func nodesToJSON(nodes []dag.Node) []map[string]interface{} {
        var out []map[string]interface{}
        for _, n := range nodes {
                out = append(out, map[string]interface{}{
                        "name":       n.Task.Name,
                        "triggered":  n.Triggered,
                        "reason":     n.TriggeredReason,
                        "repo":       n.Task.Execute.Repo,
                        "script":     n.Task.Execute.Script,
                        "image":      n.Task.Execute.Image,
                        "manual":     n.Task.Manual,
                        "depends_on": n.Task.DependsOn,
                })
        }
        return out
}

// Package web implements the HTTP webhook receiver for MetaCI.
//
// It receives push events from GitLab (and other git hosts), extracts the
// changed-file information, and passes it to the DAG engine for evaluation.
package web

import (
        "encoding/json"
        "fmt"
        "strings"
)

// --- GitLab Webhook Payload ---

// GitLabPushEvent represents the JSON payload sent by GitLab on a push event.
// Only the fields we need are extracted; unknown fields are silently ignored.
type GitLabPushEvent struct {
        ObjectKind string         `json:"object_kind"`
        Ref        string         `json:"ref"`
        Project    GitLabProject  `json:"project"`
        Commits    []GitLabCommit `json:"commits"`
}

// GitLabProject holds the project metadata from the webhook.
type GitLabProject struct {
        PathWithNamespace string `json:"path_with_namespace"`
        DefaultBranch     string `json:"default_branch"`
}

// GitLabCommit holds a single commit's file-change data.
type GitLabCommit struct {
        ID       string   `json:"id"`
        Message  string   `json:"message"`
        Added    []string `json:"added"`
        Modified []string `json:"modified"`
        Removed  []string `json:"removed"`
}

// --- Normalized Event ---

// PushEvent is the host-agnostic representation of a push event.
// After parsing, all downstream code works with PushEvent only.
type PushEvent struct {
        // Repo is the path_with_namespace (e.g. "org/service-a") or a custom
        // identifier that maps to a ci_repo in the pipeline.
        Repo string

        // Branch is the short branch name (e.g. "main", not "refs/heads/main").
        Branch string

        // FilesChanged is the deduplicated set of all files touched across all
        // commits in this push.
        FilesChanged map[string]bool

        // CommitCount is the number of commits in this push.
        CommitCount int
}

// IsEmpty returns true if the event contains no file changes.
func (e *PushEvent) IsEmpty() bool {
        return len(e.FilesChanged) == 0
}

// --- File Classification ---

// FileClassifier determines whether a file is "code" or "non-code" based on
// configurable patterns. This is used to decide whether changes should trigger
// CI tasks.
type FileClassifier struct {
        // CodePatterns is a list of file extensions considered as code.
        // Defaults to common Go, Python, Rust, JS/TS extensions.
        CodePatterns []string

        // ExcludePatterns is a list of path prefixes or extensions to always
        // exclude from triggering builds. E.g. ["docs/", ".md"].
        ExcludePatterns []string
}

// DefaultClassifier returns a classifier with sensible defaults for a
// polyglot Go/Python/Rust project.
func DefaultClassifier() *FileClassifier {
        return &FileClassifier{
                CodePatterns: []string{
                        ".go", ".py", ".rs", ".js", ".ts", ".java", ".c", ".cpp", ".h",
                        ".tf", ".yaml", ".yml", ".toml", ".json",
                },
                ExcludePatterns: []string{
                        "docs/",
                        ".md",
                        "LICENSE",
                        "CHANGELOG",
                        ".gitignore",
                },
        }
}

// IsCode returns true if the file should be considered as code (i.e., its
// change should trigger CI tasks).
func (fc *FileClassifier) IsCode(path string) bool {
        // Check exclude patterns first.
        for _, pattern := range fc.ExcludePatterns {
                if strings.HasSuffix(path, pattern) || strings.HasPrefix(path, pattern) {
                        return false
                }
        }
        // Check code patterns.
        for _, ext := range fc.CodePatterns {
                if strings.HasSuffix(path, ext) {
                        return true
                }
        }
        return false
}

// HasCodeChanges returns true if any file in the changed set is classified as code.
func (fc *FileClassifier) HasCodeChanges(files map[string]bool) bool {
        for path := range files {
                if fc.IsCode(path) {
                        return true
                }
        }
        return false
}

// ChangedRepos extracts the set of repos that have code-level changes.
// The input is a map of repo-name → set of changed files.
func (fc *FileClassifier) ChangedRepos(repoFiles map[string]map[string]bool) map[string]bool {
        result := make(map[string]bool)
        for repo, files := range repoFiles {
                result[repo] = fc.HasCodeChanges(files)
        }
        return result
}

// --- Parsing ---

// ParseGitLabPushEvent parses raw JSON from a GitLab push webhook into a
// normalized PushEvent. It only processes "push" events (object_kind == "push").
func ParseGitLabPushEvent(payload []byte) (*PushEvent, error) {
        var raw GitLabPushEvent
        if err := json.Unmarshal(payload, &raw); err != nil {
                return nil, fmt.Errorf("parse webhook JSON: %w", err)
        }

        if raw.ObjectKind != "push" {
                return nil, fmt.Errorf("unsupported event kind %q (only \"push\" is supported)", raw.ObjectKind)
        }

        // Normalize branch: "refs/heads/main" → "main"
        branch := strings.TrimPrefix(raw.Ref, "refs/heads/")
        branch = strings.TrimPrefix(branch, "refs/tags/")

        // Collect all changed files across all commits.
        files := make(map[string]bool)
        for _, c := range raw.Commits {
                for _, f := range c.Added {
                        files[f] = true
                }
                for _, f := range c.Modified {
                        files[f] = true
                }
                for _, f := range c.Removed {
                        files[f] = true
                }
        }

        return &PushEvent{
                Repo:         raw.Project.PathWithNamespace,
                Branch:       branch,
                FilesChanged: files,
                CommitCount:  len(raw.Commits),
        }, nil
}
